using System;


class Person{
    public  void work()
    {
        Console.WriteLine("Humans can work");
    }
}

class Student:Person{
    public  void study()
    {
        Console.WriteLine("students can study");
    }
}
public class HelloWorld
{
    public static void Main(string[] args)
    {
        Student s1 = new Student();
        s1.work();
        s1.study();
    }
}
