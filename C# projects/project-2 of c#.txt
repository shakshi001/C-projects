using System;

public class HelloWorld
{
    public static void Main(string[] args)
    {
       int number;
       int i = 0;
       int sum = 0;
       Console.WriteLine("Kindly , enter your number ...");
       number = Convert.ToInt32(Console.ReadLine());
       while(i<=number)
       {
           sum += i;
           i++;
       }
       
       Console.WriteLine("Your sum is : "+sum);
       
    }
}
